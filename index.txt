        一般地，函数 <span class="formula">y = a<sup>x</sup> (a &gt; 0 且 a ≠ 1)</span> 叫做指数函数，其中 <i>x</i> 是自变量，定义域为 <b>R</b>。
      </div>
      <p style="font-size:13.5px; color:var(--text-secondary);">
        <strong>形式上的三个“必须”（缺一不可）：</strong><br/>
        ① <b>底数限制</b>：<i>a</i> &gt; 0 且 <i>a</i> ≠ 1（若 <i>a</i> ≤ 0，如 (-2)<sup>1/2</sup> 在实数集无意义；若 <i>a</i> = 1，则 <i>y</i> = 1<sup><i>x</i></sup> = 1 是常函数，无研究价值）。<br/>
        ② <b>系数必须为 1</b>：<i>a<sup>x</sup></i> 前面的系数只能是 1。例如 <i>y</i> = 2·3<sup><i>x</i></sup> 只能叫“指数型函数”，不是指数函数。<br/>
        ③ <b>指数位置必须是单独的 <i>x</i></b>：例如 <i>y</i> = 2<sup><i>x</i>+1</sup>、<i>y</i> = 3<sup>2<i>x</i></sup> 都不是严格定义下的指数函数（需化简或作为复合函数处理）。
      </p>
    </div>

        ctx.restore();
      }

      // 4. Reference Curves
      if (state.showRef2) {
        plotFunction(x => Math.pow(2, x), '#059669', 1.6, true);
      }
      if (state.showRef3) {
        plotFunction(x => Math.pow(3, x), '#d97706', 1.6, true);
      }
      if (state.showReciprocal) {
        plotFunction(x => Math.pow(1 / state.a, x), '#e11d48', 1.8, true);
      }
      if (state.showLog) {
        // y = x axis of symmetry
        plotFunction(x => x, '#94a3b8', 1, true);
        // y = log_a(x)
        ctx.save();
        ctx.strokeStyle = '#7c3aed';
        ctx.lineWidth = 1.8;
        ctx.beginPath();
        let started = false;
        const startPx = Math.max(0, Math.floor(toPxX(0.002)));
        for (let px = startPx; px <= w; px += 1.5) {
          const x = toMathX(px);
          if (x <= 0) continue;
          const y = Math.log(x) / Math.log(state.a);
          if (!Number.isFinite(y) || y > maxY + 8 || y < minY - 8) {
            started = false;
            continue;
          }
          const py = toPxY(y);
          if (!started) {
            ctx.moveTo(px, py);
            started = true;
          } else {
            ctx.lineTo(px, py);
          }
        }
        ctx.stroke();
        ctx.restore();
      }

      // 5. Primary Curve: y = k * a^(x - h) + b
      const mainFn = x => state.k * Math.pow(state.a, x - state.h) + state.b;
      plotFunction(mainFn, '#2563eb', 2.8, false);

      // 6. Highlight Fixed Point (h, k + b)
      const fixPxX = toPxX(state.h);
      const fixPxY = toPxY(state.k + state.b);
      ctx.fillStyle = '#2563eb';
      ctx.beginPath();
      ctx.arc(fixPxX, fixPxY, 5, 0, Math.PI * 2);
      ctx.fill();
      ctx.strokeStyle = '#ffffff';
      ctx.lineWidth = 1.8;
      ctx.stroke();

      ctx.fillStyle = '#1e3a8a';
      ctx.font = 'bold 11.5px -apple-system, sans-serif';
      ctx.textAlign = 'left';
      ctx.fillText(`定点(${formatNum(state.h)}, ${formatNum(state.k + state.b)})`, fixPxX + 8, fixPxY - 6);

      // 7. Interactive Coordinate Probe
      if (state.probeX !== null) {
        const pyVal = mainFn(state.probeX);
        const ppx = toPxX(state.probeX);
        const ppy = toPxY(pyVal);

        ctx.save();
        ctx.setLineDash([3, 3]);
        ctx.strokeStyle = '#0f172a';
        ctx.lineWidth = 1;
        ctx.beginPath();
        ctx.moveTo(ppx, originY);
        ctx.lineTo(ppx, ppy);
        ctx.lineTo(originX, ppy);
        ctx.stroke();
        ctx.restore();

        ctx.fillStyle = '#e11d48';
        ctx.beginPath();
        ctx.arc(ppx, ppy, 5.5, 0, Math.PI * 2);
        ctx.fill();

        probeInfo.textContent = `x = ${state.probeX.toFixed(2)},  y = ${pyVal.toFixed(3)}`;
      }

      ctx.restore();
    }

    function renderQuiz() {
      const container = document.getElementById('quizContainer');

      quizData.forEach((item, idx) => {
        const card = document.createElement('div');
        card.className = 'quiz-card';

        const qEl = document.createElement('div');
        qEl.className = 'quiz-q';

        expEl.className = 'quiz-explain';
        expEl.innerHTML = `<strong>解析：</strong>${item.explain}`;

        item.options.forEach((optText, oIdx) => {
          const btn = document.createElement('button');
          btn.className = 'opt-btn';
          btn.textContent = optText;
          btn.addEventListener('click', () => {
            if (userAnswers[idx] !== null) return;
            userAnswers[idx] = oIdx;
            answeredCount++;
            if (oIdx === item.answer) {
              score += 10;
              btn.classList.add('correct');
            } else {
              btn.classList.add('wrong');
              optGrid.children[item.answer].classList.add('correct');
            }
            expEl.classList.add('show');

            document.getElementById('qScore').textContent = score;
          });
          optGrid.appendChild(btn);
        });

        card.appendChild(optGrid);
        card.appendChild(expEl);
        container.appendChild(card);
      });
    }
